    id INT AUTO_INCREMENT PRIMARY KEY,
    nombre char(120) NOT NULL,

CREATE TABLE pajaros (
    id INT AUTO_INCREMENT PRIMARY KEY,
    nombre VARCHAR(50) NOT NULL,
    raza VARCHAR(50) NOT NULL UNIQUE
);



CREATE TABLE avistamientos (

    id INT AUTO_INCREMENT PRIMARY KEY,

    usuario_id INT NOT NULL,
    pajaro_id INT NOT NULL,

    lugar POINT NOT NULL,

    fecha_avistamiento DATETIME NOT NULL,

    FOREIGN KEY (usuario_id) REFERENCES usuarios(id) ON DELETE RESTRICT,
    FOREIGN KEY (pajaro_id) REFERENCES pajaros(id) ON DELETE CASCADE,

    SPATIAL INDEX(lugar)
);



DELIMITER //

CREATE TRIGGER before_delete_usuario

BEFORE DELETE ON usuarios

FOR EACH ROW

BEGIN

    UPDATE avistamientos

    SET usuario_id = 1

    WHERE usuario_id = OLD.id;

END;

//

DELIMITER ;
